cmake_minimum_required(VERSION 3.21)
project(ffmpeg-tester LANGUAGES C CXX)

# One external discovery entry per configure/build tree. The same source
# project is used by the vcpkg Port and standalone SDK consumers.
set(FFBUILD_TESTER_MODE CONFIG CACHE STRING "CONFIG, MODULE, LEGACY, PKGCONFIG or COMPONENT")
set(FFBUILD_TESTER_PREFIX "" CACHE PATH "Explicit SDK prefix for standalone consumers")
set(FFBUILD_TESTER_MODULES "" CACHE STRING "Expected modules from the build plan/inventory")
option(FFBUILD_TESTER_NATIVE "Require a complete installed Release+Debug SDK" ON)
option(FFBUILD_TESTER_DEPLOY "Deploy imported DLLs for standalone consumers" OFF)
option(FFBUILD_TESTER_CRT_MATRIX "Add C/C++ static and dynamic CRT probes" OFF)
option(FFBUILD_TESTER_FULL_ONLY "Build only the comprehensive consumer" OFF)

set(_core avutil avcodec avformat avdevice avfilter swresample swscale)
set(_known ${_core} avresample postproc)
if(NOT FFBUILD_TESTER_MODE MATCHES "^(CONFIG|MODULE|LEGACY|PKGCONFIG|COMPONENT)$")
    message(FATAL_ERROR "Unknown Tester discovery mode: ${FFBUILD_TESTER_MODE}")
endif()
if(NOT FFBUILD_TESTER_PREFIX)
    if(DEFINED VCPKG_INSTALLED_DIR AND DEFINED VCPKG_TARGET_TRIPLET)
        set(FFBUILD_TESTER_PREFIX "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}")
    elseif(DEFINED _VCPKG_INSTALLED_DIR AND DEFINED VCPKG_TARGET_TRIPLET)
        set(FFBUILD_TESTER_PREFIX "${_VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}")
    elseif(CMAKE_PREFIX_PATH AND IS_DIRECTORY "${CMAKE_PREFIX_PATH}/share/ffmpeg")
        set(FFBUILD_TESTER_PREFIX "${CMAKE_PREFIX_PATH}")
    else()
        message(FATAL_ERROR "Tester requires the selected SDK prefix")
    endif()
endif()
file(REAL_PATH "${FFBUILD_TESTER_PREFIX}" _prefix)
# Independent inventory, not the imported target set. Production verifies
# requested pkg-config modules before generating the package metadata.
file(GLOB _pcs RELATIVE "${_prefix}/lib/pkgconfig" "${_prefix}/lib/pkgconfig/lib*.pc")
set(_modules "")
foreach(_pc IN LISTS _pcs)
    if(NOT _pc MATCHES "^lib([a-z0-9_]+)[.]pc$")
        if(FFBUILD_TESTER_NATIVE)
            continue() # The full vcpkg prefix also contains vendor .pc files.
        endif()
        message(FATAL_ERROR "Invalid FFmpeg module inventory: ${_pc}")
    endif()
    set(_candidate "${CMAKE_MATCH_1}")
    if(_candidate IN_LIST _known)
        list(APPEND _modules "${_candidate}")
    elseif(NOT FFBUILD_TESTER_NATIVE)
        message(FATAL_ERROR "Unexpected public SDK pkg-config module: ${_candidate}")
    endif()
endforeach()
foreach(_m IN LISTS _core FFBUILD_TESTER_MODULES)
    if(NOT _m IN_LIST _modules)
        message(FATAL_ERROR "Expected SDK module missing: ${_m}")
    endif()
endforeach()
if(FFBUILD_TESTER_MODULES)
    set(_expected "${FFBUILD_TESTER_MODULES}")
    list(SORT _expected)
    list(SORT _modules)
    if(NOT _expected STREQUAL _modules)
        message(FATAL_ERROR "SDK module inventory differs from expected modules")
    endif()
endif()
set(CMAKE_FIND_USE_PACKAGE_REGISTRY FALSE)
set(CMAKE_FIND_USE_SYSTEM_PACKAGE_REGISTRY FALSE)
set(_runtime_before "${CMAKE_MSVC_RUNTIME_LIBRARY}")
if(FFBUILD_TESTER_MODE STREQUAL "PKGCONFIG")
    set(PKG_CONFIG_USE_CMAKE_PREFIX_PATH FALSE)
    find_package(PkgConfig REQUIRED)
    set(_pc_all "")
    foreach(_m IN LISTS _modules)
        list(APPEND _pc_all "lib${_m}")
    endforeach()
    pkg_check_modules(FFALL REQUIRED IMPORTED_TARGET ${_pc_all})
    set(_pc_rows "full|$<TARGET_PROPERTY:PkgConfig::FFALL,INTERFACE_LINK_LIBRARIES>\n")
else()
    if(FFBUILD_TESTER_MODE STREQUAL "COMPONENT")
        find_package(FFmpeg CONFIG REQUIRED COMPONENTS avformat)
    elseif(FFBUILD_TESTER_MODE STREQUAL "CONFIG")
        # No Module call and no FFMPEG_ROOT override in this configure.
        find_package(FFmpeg CONFIG REQUIRED)
    else()
        list(PREPEND CMAKE_MODULE_PATH "${_prefix}/share/ffmpeg")
        find_package(FFmpeg MODULE REQUIRED)
    endif()
    if(NOT "${_runtime_before}" STREQUAL "${CMAKE_MSVC_RUNTIME_LIBRARY}")
        message(FATAL_ERROR "FFmpeg changed the consumer CRT")
    endif()
    if(FFBUILD_TESTER_MODE MATCHES "^(CONFIG|COMPONENT)$")
        file(REAL_PATH "${FFmpeg_DIR}" _actual_package)
        file(REAL_PATH "${_prefix}/share/ffmpeg" _expected_package)
        if(NOT _actual_package STREQUAL _expected_package)
            message(FATAL_ERROR "Config discovered a different SDK")
        endif()
    endif()
    set(_actual "${FFMPEG_MODULES}")
    list(SORT _actual)
    list(SORT _modules)
    if(NOT _actual STREQUAL _modules)
        message(FATAL_ERROR "CMake targets differ from installed module inventory")
    endif()
    set(_selected "")
    foreach(_m IN LISTS _modules)
        if(NOT TARGET FFmpeg::${_m})
            message(FATAL_ERROR "Missing imported FFmpeg module: ${_m}")
        endif()
        if(FFBUILD_TESTER_NATIVE AND
           "${FFMPEG_lib${_m}_LIBRARY_DEBUG}" STREQUAL "${FFMPEG_lib${_m}_LIBRARY_RELEASE}")
            message(FATAL_ERROR "Native FFmpeg Tester: ${_m} Debug library falls back to Release")
        endif()
        string(APPEND _selected "${_m}|$<TARGET_FILE:FFmpeg::${_m}>|$<TARGET_LINKER_FILE:FFmpeg::${_m}>\n")
    endforeach()
    file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/selected-$<CONFIG>.txt" CONTENT "${_selected}")
endif()

set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(_src "${CMAKE_CURRENT_BINARY_DIR}/src")
file(MAKE_DIRECTORY "${_src}")
# Small public API probes only. The registry calls below do not require
# codecs, media, devices, network, or an enabled optional feature.
function(probe_source out name modules)
    set(headers "")
    set(checks "")
    foreach(m IN LISTS modules)
        set(header "${m}.h")
        if(m STREQUAL "postproc")
            set(header "postprocess.h")
        endif()
        string(APPEND headers "#include <lib${m}/${header}>\n")
        if(ARGN AND NOT m IN_LIST ARGN)
            continue()
        endif()
        string(APPEND checks "    if (${m}_version() == 0) return 1;\n")
        if(m STREQUAL "avutil")
            string(APPEND headers "#include <libavutil/mem.h>\n")
            string(APPEND checks "    { void *p = av_malloc(32); if (!p) return 2; av_free(p); }\n")
        elseif(m STREQUAL "avcodec")
            string(APPEND checks "#if LIBAVCODEC_VERSION_MAJOR < 58\n    avcodec_register_all();\n#endif\n    (void)avcodec_find_decoder(AV_CODEC_ID_NONE);\n    { AVCodecContext *c = avcodec_alloc_context3(NULL); if (!c) return 2; avcodec_free_context(&c); }\n")
        elseif(m STREQUAL "avformat")
            string(APPEND checks "#if LIBAVFORMAT_VERSION_MAJOR < 58\n    av_register_all();\n#endif\n    (void)av_find_input_format(\"\");\n    { AVFormatContext *c = avformat_alloc_context(); if (!c) return 2; avformat_free_context(c); }\n")
        elseif(m STREQUAL "avdevice")
            string(APPEND checks "    avdevice_register_all();\n")
        elseif(m STREQUAL "avfilter")
            string(APPEND checks "#if LIBAVFILTER_VERSION_MAJOR < 7\n    avfilter_register_all();\n#endif\n    (void)avfilter_get_by_name(\"\");\n    { AVFilterGraph *g = avfilter_graph_alloc(); if (!g) return 2; avfilter_graph_free(&g); }\n")
        endif()
    endforeach()
    set(source "#include <stddef.h>\n#ifdef __cplusplus\n#ifndef __STDC_CONSTANT_MACROS\n#define __STDC_CONSTANT_MACROS\n#endif\nextern \"C\" {\n#endif\n${headers}#ifdef __cplusplus\n}\n#endif\n#if defined(EXPECT_DLL_CRT) && (!defined(_MT) || (defined(_DLL) != EXPECT_DLL_CRT) || (defined(_DEBUG) != EXPECT_DEBUG_CRT))\n#error Consumer CRT does not match the requested runtime\n#endif\nint main(void) {\n${checks}    return 0;\n}\n")
    file(WRITE "${_src}/${name}.c" "${source}")
    file(WRITE "${_src}/${name}.cpp" "${source}")
    set(${out} "${_src}/${name}" PARENT_SCOPE)
endfunction()

function(add_probe target source language)
    add_executable(${target} "${source}.${language}")
    set_target_properties(${target} PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>/${target}")
    if(FFBUILD_TESTER_DEPLOY AND NOT FFBUILD_TESTER_MODE MATCHES "^(LEGACY|PKGCONFIG)$")
        add_custom_command(TARGET ${target} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different
                "$<TARGET_RUNTIME_DLLS:${target}>" "$<TARGET_FILE_DIR:${target}>"
            COMMAND_EXPAND_LISTS)
    endif()
endfunction()

# A separate Config COMPONENTS request must link its actual dependency
# chain without calling find_package again or linking the aggregate target.
if(FFBUILD_TESTER_MODE STREQUAL "COMPONENT")
    probe_source(component_source component "avformat;avcodec;avutil")
    add_probe(check-component-avformat "${component_source}" c)
    target_link_libraries(check-component-avformat PRIVATE FFmpeg::avformat)
    return()
endif()

function(link_full target)
    if(FFBUILD_TESTER_MODE STREQUAL "PKGCONFIG")
        target_link_libraries(${target} PRIVATE PkgConfig::FFALL)
    elseif(FFBUILD_TESTER_MODE STREQUAL "LEGACY")
        target_include_directories(${target} PRIVATE ${FFMPEG_INCLUDE_DIRS})
        target_link_libraries(${target} PRIVATE ${FFMPEG_LIBRARIES})
    else()
        target_link_libraries(${target} PRIVATE FFmpeg::FFmpeg)
    endif()
endfunction()

probe_source(full_source full "${_modules}")
add_probe(check-full "${full_source}" c)
link_full(check-full)
probe_source(cpp_source cpp "${_modules}" "avutil")
add_probe(check-full-cpp "${cpp_source}" cpp)
link_full(check-full-cpp)

if(FFBUILD_TESTER_MODE STREQUAL "LEGACY")
    set(_legacy_rows "")
    set(_legacy_includes "")
    foreach(t check-full check-full-cpp)
        string(APPEND _legacy_rows "${t}|$<TARGET_GENEX_EVAL:${t},$<TARGET_PROPERTY:${t},LINK_LIBRARIES>>\n")
        string(APPEND _legacy_includes "${t}|$<TARGET_GENEX_EVAL:${t},$<TARGET_PROPERTY:${t},INCLUDE_DIRECTORIES>>\n")
    endforeach()
    file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/legacy-$<CONFIG>.txt" CONTENT "${_legacy_rows}")
    file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/legacy-includes-$<CONFIG>.txt" CONTENT "${_legacy_includes}")
    # Cross-check the raw Legacy dependency closure against the imported
    # modules' configuration-specific transitive interface. Do not use
    # that interface to link the Legacy probes themselves.
    set(_reference_rows "")
    foreach(m IN LISTS _modules)
        string(APPEND _reference_rows "${m}|$<TARGET_LINKER_FILE:FFmpeg::${m}>;$<TARGET_GENEX_EVAL:FFmpeg::${m},$<TARGET_PROPERTY:FFmpeg::${m},INTERFACE_LINK_LIBRARIES>>\n")
    endforeach()
    file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/legacy-reference-$<CONFIG>.txt" CONTENT "${_reference_rows}")
endif()

if(NOT FFBUILD_TESTER_FULL_ONLY AND NOT FFBUILD_TESTER_MODE STREQUAL "LEGACY")
    foreach(m IN LISTS _modules)
        # Link only the requested target, but reference avutil through its
        # published dependency graph for all modules that depend on it.
        set(probe_modules "${m}")
        if(NOT FFBUILD_TESTER_MODE STREQUAL "PKGCONFIG" AND NOT m MATCHES "^(avutil|avresample|postproc)$")
            list(APPEND probe_modules avutil)
        endif()
        probe_source(source "${m}" "${probe_modules}")
        add_probe(check-${m} "${source}" c)
        if(FFBUILD_TESTER_MODE STREQUAL "PKGCONFIG")
            pkg_check_modules(FF_${m} REQUIRED IMPORTED_TARGET "lib${m}")
            # pkg-config private dependencies are not promised to propagate
            # through ordinary --libs in a shared SDK. Use its actual public
            # flags, not a manually added avutil import library.
            target_link_libraries(check-${m} PRIVATE PkgConfig::FF_${m})
            string(APPEND _pc_rows "${m}|$<TARGET_PROPERTY:PkgConfig::FF_${m},INTERFACE_LINK_LIBRARIES>\n")
        else()
            target_link_libraries(check-${m} PRIVATE FFmpeg::${m})
        endif()
    endforeach()
endif()

if(FFBUILD_TESTER_CRT_MATRIX)
    if(NOT FFBUILD_TESTER_MODE STREQUAL "CONFIG" OR NOT MSVC)
        message(FATAL_ERROR "CRT matrix requires an MSVC Config-only consumer")
    endif()
    probe_source(crt_source crt "avcodec;avutil")
    foreach(lang c cpp)
        foreach(crt MT MD)
            string(TOLOWER "${crt}" rt)
            set(t "check-crt-${lang}-${rt}")
            add_probe(${t} "${crt_source}" "${lang}")
            if(crt STREQUAL "MD")
                set(dll 1)
                set(runtime "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
            else()
                set(dll 0)
                set(runtime "MultiThreaded$<$<CONFIG:Debug>:Debug>")
            endif()
            set_property(TARGET ${t} PROPERTY MSVC_RUNTIME_LIBRARY "${runtime}")
            target_compile_definitions(${t} PRIVATE EXPECT_DLL_CRT=${dll} EXPECT_DEBUG_CRT=$<CONFIG:Debug>)
            target_link_libraries(${t} PRIVATE FFmpeg::avcodec)
        endforeach()
    endforeach()
endif()

    file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/pkgconfig-$<CONFIG>.txt" CONTENT "${_pc_rows}")
endif()
