cmake_minimum_required(VERSION 3.21)
project(ffmpeg_archive_consumer C)
list(APPEND CMAKE_MODULE_PATH "${FFMPEG_ROOT}/share/ffmpeg")
find_package(FFmpeg MODULE REQUIRED)
add_executable(check-full main.c)
target_link_libraries(check-full PRIVATE FFmpeg::FFmpeg)
add_executable(check-demand main.c)
target_link_libraries(check-demand PRIVATE FFmpeg::avformat)
add_executable(check-legacy main.c)
target_include_directories(check-legacy PRIVATE ${FFMPEG_INCLUDE_DIRS})
target_link_libraries(check-legacy PRIVATE ${FFMPEG_LIBRARIES})
foreach(target check-full check-demand check-legacy)
  set_target_properties(${target} PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>/${target}")
endforeach()
foreach(target check-full check-demand)
  add_custom_command(TARGET ${target} POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "$<TARGET_RUNTIME_DLLS:${target}>" "$<TARGET_FILE_DIR:${target}>"
    COMMAND_EXPAND_LISTS)
endforeach()
set(selected "")
foreach(module avutil avcodec avformat avfilter avdevice swresample swscale)
  if(NOT TARGET FFmpeg::${module})
    message(FATAL_ERROR "Missing SDK module: ${module}")
  endif()
  string(APPEND selected "$<TARGET_FILE:FFmpeg::${module}>|$<TARGET_LINKER_FILE:FFmpeg::${module}>\n")
endforeach()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/selected-$<CONFIG>.txt" CONTENT "${selected}")
